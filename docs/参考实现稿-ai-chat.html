好的！这是一个基于 Pandas 的 RFM 用户价值分层模型代码框架：

# rfm_model.py
import pandas as pd
import numpy as np

def build_rfm_model(df, user_col='user_id',
              date_col='order_date',
              amount_col='amount'):

  # 计算最近一次购买距今天数（Recency）
  now = df[date_col].max()
  rfm = df.groupby(user_col).agg({
    'Recency': lambda x: (now - x.max()).days,
    'Frequency': 'count',
    'Monetary': amount_col
  })

  # 五分位数打分（1-5 分，5 最高）
  r_labels = range(5, 0, -1)
  for col in rfm.columns:
    rfm[f'{col}_Score'] = pd.qcut(rfm[col], 5,
      labels=r_labels).astype(int)

  # 计算 RFM 总分并分段
  rfm['RFM_Score'] = (rfm[[...]] + '_Score']).sum(axis=1)
